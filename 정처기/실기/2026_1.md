```c
#include <stdio.h>
double arr(int *p, int len) {
    double av = 0.0;
    for (int i = 0; i < len; i++) {
        av += (double)p[i];
    }
    return av / len;
}
double arrv(int *p, int len) {
    double av = 0.0;
    for (int i = 0; i < len; i++) {
        av += (double)(*(p + i));
    }
    return av / len;
}
int main() {
    int score[] = {85, 90, 25, 30, 55, 45, 95, 55, 10, 40};
    int len = 10;
    printf("%.2f\n", arr(score, len) + arrv(score, len));
    return 0;
}
```
